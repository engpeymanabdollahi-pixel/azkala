# ============================================================
# 📊 Azkala Project - Comprehensive Data Collection Script
# ============================================================
# این اسکریپت اطلاعات واقعی پروژه را جمع‌آوری می‌کند
# خروجی در فایل azkala_report_data.txt ذخیره می‌شود

$OutputFile = "azkala_report_data.txt"
$BackendPath = "backend"
$FrontendPath = "frontend"

# پاک کردن فایل قبلی
if (Test-Path $OutputFile) { Remove-Item $OutputFile }

# تابع برای نوشتن در فایل و کنسول
function Write-Section($title) {
    $separator = "=" * 70
    "`n$separator`n$title`n$separator" | Tee-Object -FilePath $OutputFile -Append
}

function Write-Data($data) {
    $data | Tee-Object -FilePath $OutputFile -Append
}

# ============================================================
# بخش ۱: آمار کلی پروژه
# ============================================================
Write-Section "📊 SECTION 1: PROJECT OVERVIEW"
Write-Data "Date: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"
Write-Data "Project Path: $(Get-Location)"

# ============================================================
# بخش ۲: آمار Backend (PHP)
# ============================================================
Write-Section "🔧 SECTION 2: BACKEND STATISTICS"

# تعداد فایل‌های PHP
$phpFiles = Get-ChildItem -Path $BackendPath -Filter "*.php" -Recurse -File
Write-Data "Total PHP Files: $($phpFiles.Count)"
Write-Data "Total PHP Lines: $(($phpFiles | Get-Content | Measure-Object -Line).Lines)"

# دسته‌بندی فایل‌های Backend
$categories = @{
    "Controllers" = "app/Http/Controllers"
    "Services" = "app/Services"
    "Repositories" = "app/Repositories"
    "Models" = "app/Models"
    "DTOs" = "app/DTOs"
    "Migrations" = "database/migrations"
    "Factories" = "database/factories"
    "Tests" = "tests"
}

foreach ($cat in $categories.GetEnumerator()) {
    $path = Join-Path $BackendPath $cat.Value
    if (Test-Path $path) {
        $files = Get-ChildItem -Path $path -Filter "*.php" -Recurse -File
        $lines = ($files | Get-Content | Measure-Object -Line).Lines
        Write-Data "$($cat.Key): $($files.Count) files, $lines lines"
    }
}

# ============================================================
# بخش ۳: آمار Frontend (TypeScript/React)
# ============================================================
Write-Section "🎨 SECTION 3: FRONTEND STATISTICS"

$tsFiles = Get-ChildItem -Path $FrontendPath -Include "*.ts","*.tsx" -Recurse -File | Where-Object { $_.FullName -notmatch "node_modules" }
Write-Data "Total TS/TSX Files: $($tsFiles.Count)"
Write-Data "Total TS Lines: $(($tsFiles | Get-Content | Measure-Object -Line).Lines)"

# دسته‌بندی Frontend
$feCategories = @{
    "Pages" = "src/pages"
    "Components" = "src/components"
    "Features" = "src/features"
    "Stores" = "src/stores"
    "Services" = "src/services"
    "Hooks" = "src/hooks"
}

foreach ($cat in $feCategories.GetEnumerator()) {
    $path = Join-Path $FrontendPath $cat.Value
    if (Test-Path $path) {
        $files = Get-ChildItem -Path $path -Include "*.ts","*.tsx" -Recurse -File
        $lines = ($files | Get-Content | Measure-Object -Line).Lines
        Write-Data "$($cat.Key): $($files.Count) files, $lines lines"
    }
}

# ============================================================
# بخش ۴: نتایج تست Backend
# ============================================================
Write-Section "🧪 SECTION 4: BACKEND TEST RESULTS"
Set-Location $BackendPath
try {
    $testOutput = php artisan test --stop-on-failure 2>&1
    Write-Data $testOutput
} catch {
    Write-Data "Error running tests: $_"
}
Set-Location ..

# ============================================================
# بخش ۵: ساختار دیتابیس
# ============================================================
Write-Section "🗄️ SECTION 5: DATABASE STRUCTURE"

# لیست migrationها
$migrations = Get-ChildItem -Path "$BackendPath/database/migrations" -Filter "*.php" | Sort-Object Name
Write-Data "Total Migrations: $($migrations.Count)"
Write-Data "Migration List:"
$migrations | ForEach-Object { Write-Data "  - $($_.Name)" }

# ============================================================
# بخش ۶: لیست سرویس‌ها و Repositoryها
# ============================================================
Write-Section "📦 SECTION 6: SERVICES & REPOSITORIES"

Write-Data "`nServices:"
Get-ChildItem -Path "$BackendPath/app/Services" -Filter "*.php" -Recurse | ForEach-Object {
    $lines = (Get-Content $_.FullName | Measure-Object -Line).Lines
    Write-Data "  - $($_.Name): $lines lines"
}

Write-Data "`nRepositories:"
Get-ChildItem -Path "$BackendPath/app/Repositories" -Filter "*.php" -Recurse | ForEach-Object {
    $lines = (Get-Content $_.FullName | Measure-Object -Line).Lines
    Write-Data "  - $($_.Name): $lines lines"
}

# ============================================================
# بخش ۷: لیست کنترلرها
# ============================================================
Write-Section "🎮 SECTION 7: CONTROLLERS"

Get-ChildItem -Path "$BackendPath/app/Http/Controllers" -Filter "*.php" -Recurse | ForEach-Object {
    $lines = (Get-Content $_.FullName | Measure-Object -Line).Lines
    $relativePath = $_.FullName.Replace((Get-Location).Path + "\$BackendPath\", "")
    Write-Data "  - $relativePath`: $lines lines"
}

# ============================================================
# بخش ۸: وضعیت Frontend Build
# ============================================================
Write-Section "📦 SECTION 8: FRONTEND BUILD STATUS"

Set-Location $FrontendPath
try {
    # بررسی node_modules
    if (Test-Path "node_modules") {
        Write-Data "node_modules: EXISTS"
    } else {
        Write-Data "node_modules: MISSING"
    }
    
    # بررسی dist
    if (Test-Path "dist") {
        Write-Data "dist folder: EXISTS"
        $distSize = (Get-ChildItem -Path "dist" -Recurse | Measure-Object -Property Length -Sum).Sum / 1MB
        Write-Data "dist size: $([math]::Round($distSize, 2)) MB"
        
        # لیست فایل‌های dist
        Get-ChildItem -Path "dist/assets" -File | ForEach-Object {
            $size = [math]::Round($_.Length / 1KB, 2)
            Write-Data "  - $($_.Name): $size KB"
        }
    } else {
        Write-Data "dist folder: MISSING (needs build)"
    }
} catch {
    Write-Data "Error checking frontend: $_"
}
Set-Location ..

# ============================================================
# بخش ۹: فایل‌های تغییر یافته اخیر (Git)
# ============================================================
Write-Section "📝 SECTION 9: RECENT CHANGES (GIT)"

try {
    $gitStatus = git status --short 2>&1
    if ($gitStatus) {
        Write-Data "Git Status:"
        Write-Data $gitStatus
    } else {
        Write-Data "No uncommitted changes"
    }
    
    Write-Data "`nLast 10 commits:"
    git log --oneline -10 2>&1 | ForEach-Object { Write-Data "  $_" }
} catch {
    Write-Data "Git not available or not a git repository"
}

# ============================================================
# بخش ۱۰: فایل‌های کلیدی پیکربندی
# ============================================================
Write-Section "⚙️ SECTION 10: CONFIGURATION FILES"

$configFiles = @(
    "$BackendPath/.env",
    "$BackendPath/composer.json",
    "$FrontendPath/package.json",
    "$FrontendPath/vite.config.ts"
)

foreach ($file in $configFiles) {
    if (Test-Path $file) {
        Write-Data "`n$file`: EXISTS"
        # فقط اطلاعات کلیدی (بدون مقادیر حساس)
        if ($file -match ".env") {
            Write-Data "  (Content hidden for security)"
        } else {
            $content = Get-Content $file -Raw
            Write-Data "  Size: $($content.Length) characters"
        }
    } else {
        Write-Data "`n$file`: MISSING"
    }
}

# ============================================================
# پایان
# ============================================================
Write-Section "✅ DATA COLLECTION COMPLETE"
Write-Data "Output saved to: $OutputFile"
Write-Data "Please share this file with the architect for final report completion."
