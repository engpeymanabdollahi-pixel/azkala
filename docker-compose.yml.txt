# docker-compose.yml
services:
  # سرویس دیتابیس
  db:
    image: mysql:8.0
    container_name: azkala_db
    restart: unless-stopped
    environment:
      MYSQL_DATABASE: azkala
      MYSQL_ROOT_PASSWORD: root
      MYSQL_USER: azkala_user
      MYSQL_PASSWORD: azkala_password
    ports:
      - "3306:3306"
    volumes:
      - db_data:/var/lib/mysql

  # سرویس بک‌اند (Laravel)
  app:
    build:
      context: ./backend
      dockerfile: Dockerfile
    container_name: azkala_app
    restart: unless-stopped
    working_dir: /var/www/html
    volumes:
      - ./backend:/var/www/html
    environment:
      - DB_CONNECTION=mysql
      - DB_HOST=db
      - DB_PORT=3306
      - DB_DATABASE=azkala
      - DB_USERNAME=azkala_user
      - DB_PASSWORD=azkala_password
    depends_on:
      - db

  # سرویس فرانت‌اند (React + Nginx)
  web:
    build:
      context: ./frontend
      dockerfile: Dockerfile
    container_name: azkala_web
    restart: unless-stopped
    ports:
      - "8080:80" # دسترسی به فرانت‌اند از طریق http://localhost:8080
    depends_on:
      - app

  # سرویس ردیس (برای کش و صف‌ها)
  redis:
    image: redis:alpine
    container_name: azkala_redis
    restart: unless-stopped
    ports:
      - "6379:6379"

volumes:
  db_data:
